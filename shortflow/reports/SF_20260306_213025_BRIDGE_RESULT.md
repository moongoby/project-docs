with open('.env') as f:
    for line in f:
        line = line.strip()
        if line and not line.startswith('#') and '=' in line:
            k, _, v = line.partition('=')
            k = k.strip()
            v = v.strip()
            if k and not os.environ.get(k):
                os.environ[k] = v
