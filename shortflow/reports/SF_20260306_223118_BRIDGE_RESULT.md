`db/migrations/002_platform_accounts.sql` 이미 존재 (SF-T022에서 생성된 더 완전한 버전). 내용:

```sql
-- SF-T022: 멀티플랫폼 계정 등록 — 002_platform_accounts.sql
-- 실행: Supabase SQL Editor 또는 psql 직접 실행

-- handle_updated_at 트리거 함수가 없으면 생성
CREATE OR REPLACE FUNCTION handle_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = now();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- platform_accounts 테이블 생성
CREATE TABLE IF NOT EXISTS platform_accounts (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    platform VARCHAR(30) NOT NULL,
    account_nickname VARCHAR(100) NOT NULL,
    platform_user_id VARCHAR(200),
    platform_username VARCHAR(200),
    connected_email VARCHAR(200),
    oauth_token_file VARCHAR(300),
    auth_status VARCHAR(20) DEFAULT 'pending',
    token_expires_at TIMESTAMPTZ,
    last_upload_at TIMESTAMPTZ,
    upload_count_monthly INT DEFAULT 0,
    platform_config JSONB DEFAULT '{}',
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now()
);

CREATE UNIQUE INDEX IF NOT EXISTS idx_user_platform_nickname
    ON platform_accounts(user_id, platform, account_nickname);
CREATE INDEX IF NOT EXISTS idx_platform ON platform_accounts(platform);
CREATE INDEX IF NOT EXISTS idx_user_active ON platform_accounts(user_id, is_active);

CREATE TRIGGER set_platform_accounts_updated_at
    BEFORE UPDATE ON platform_accounts
    FOR EACH ROW EXECUTE FUNCTION handle_updated_at();

ALTER TABLE platform_accounts ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Users manage own accounts" ON platform_accounts;
CREATE POLICY "Users manage own accounts" ON platform_accounts
    FOR ALL USING (auth.uid() = user_id);

COMMENT ON TABLE platform_accounts IS 'SF-T022: 멀티플랫폼 소셜 계정 등록 및 관리';
```
