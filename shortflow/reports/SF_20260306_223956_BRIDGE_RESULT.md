**생성 파일:** `/data/shortflow/scripts/supabase_db_exec.py`

```python
import os, sys
from dotenv import load_dotenv
load_dotenv('/data/shortflow/.env')

# 방법1: DATABASE_URL 직접 사용
db_url = os.getenv('DATABASE_URL')
if not db_url:
    # 방법2: SUPABASE_URL에서 추출
    supa_url = os.getenv('SUPABASE_URL', '')
    ref = supa_url.replace('https://','').split('.')[0] if supa_url else ''
    db_pass = os.getenv('SUPABASE_DB_PASSWORD') or os.getenv('POSTGRES_PASSWORD') or os.getenv('DB_PASSWORD')
    if ref and db_pass:
        db_url = f"postgresql://postgres.{ref}:{db_pass}@aws-0-ap-northeast-2.pooler.supabase.com:6543/postgres"
    else:
        print(f"DB_URL_MISSING: ref={ref}, password={'SET' if db_pass else 'NONE'}")
        print("CEO에게 Supabase Dashboard > Settings > Database > Connection string 요청 필요")
        for k in ['SUPABASE_URL','SUPABASE_KEY','SUPABASE_SERVICE_KEY']:
            v = os.getenv(k,'')
            print(f"  {k}: {v[:4]}...({len(v)} chars)")
        sys.exit(1)

import psycopg2
try:
    conn = psycopg2.connect(db_url)
    cur = conn.cursor()
    cur.execute("SELECT version();")
    print(f"CONNECTION_OK: {cur.fetchone()[0]}")
    conn.close()
    print("DB_URL=" + db_url[:30] + "...")
except Exception as e:
    print(f"CONNECTION_FAILED: {e}")
    sys.exit(1)

conn = psycopg2.connect(db_url)
conn.autocommit = True
cur = conn.cursor()
with open('/data/shortflow/db/migrations/001_saas_schema.sql', 'r') as f:
    sql = f.read()
cur.execute(sql)
print("SCHEMA_EXEC_OK")
cur.execute("SELECT table_name FROM information_schema.tables WHERE table_schema='public' ORDER BY table_name;")
tables = [r[0] for r in cur.fetchall()]
print(f"TABLES_CREATED: {len(tables)} -> {tables}")
conn.close()
```
