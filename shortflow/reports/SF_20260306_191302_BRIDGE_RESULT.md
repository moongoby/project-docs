## 2. engine/qa_score_engine.py 생성 (v2)

파일 경로: `/data/shortflow/engine/qa_score_engine.py`

기존 v1 (720줄, ffmpeg/LUFS/TF-IDF/Gemini 복합) → v2로 교체 (단순 4항목 25점 체계)

### 생성된 코드 전문

```python
#!/usr/bin/env python3
"""SF-T017: QA Score Engine v2 — 4항목 25점 만점, 합계 85점 이상 통과"""
import json, logging, subprocess
from pathlib import Path

logger = logging.getLogger(__name__)

class QAScoreEngine:
    PASS_THRESHOLD = 85

    def __init__(self):
        self.scores = {}

    def score_script(self, script_json: dict) -> int:
        """대본 품질 (25점 만점)"""
        s = 0
        if script_json.get('hook'): s += 7          # 후크 존재
        segs = script_json.get('segments', [])
        if len(segs) >= 3: s += 6                    # 세그먼트 3개 이상
        if script_json.get('cta'): s += 4            # CTA 존재
        if script_json.get('loop_bridge') or script_json.get('loop_ending'): s += 4  # 루프엔딩
        titles = script_json.get('title_options', [])
        if titles and len(titles[0]) <= 50: s += 4   # 제목 50자 이하
        self.scores['script'] = min(s, 25)
        return self.scores['script']

    def score_video(self, video_path: str) -> int:
        """영상 품질 (25점 만점) — ffprobe 기반"""
        s = 0
        try:
            cmd = ['ffprobe', '-v', 'quiet', '-print_format', 'json', '-show_streams', '-show_format', video_path]
            result = subprocess.run(cmd, capture_output=True, text=True, timeout=30)
            info = json.loads(result.stdout)
            streams = info.get('streams', [])
            fmt = info.get('format', {})

            for st in streams:
                if st.get('codec_type') == 'video':
                    w, h = int(st.get('width',0)), int(st.get('height',0))
                    if w == 1080 and h == 1920: s += 8   # 정확한 해상도
                    fps = eval(st.get('r_frame_rate', '0/1'))
                    if 29 <= fps <= 30: s += 4            # 29.97fps
                    if st.get('codec_name') == 'h264': s += 4  # H.264
                if st.get('codec_type') == 'audio':
                    if st.get('codec_name') == 'aac': s += 4   # AAC

            duration = float(fmt.get('duration', 0))
            if 15 <= duration <= 60: s += 5              # 15~60초
        except Exception as e:
            logger.error(f"ffprobe 실패: {e}")
        self.scores['video'] = min(s, 25)
        return self.scores['video']

    def score_audio(self, script_json: dict, video_path: str = None) -> int:
        """오디오/TTS 품질 (25점 만점)"""
        s = 0
        # TTS 생성 여부 (대본 세그먼트 수 기반 추정)
        segs = script_json.get('segments', [])
        if segs: s += 10                               # TTS 대본 존재
        total_dur = script_json.get('total_duration_sec', 0)
        if 30 <= total_dur <= 55: s += 8               # 적정 길이
        elif 15 <= total_dur <= 60: s += 5
        # bg_keyword 존재 (배경음 매칭 가능)
        has_bg = all(seg.get('bg_keyword') for seg in segs)
        if has_bg: s += 7                              # 모든 세그먼트에 배경 키워드
        self.scores['audio'] = min(s, 25)
        return self.scores['audio']

    def score_metadata(self, title: str, description: str, tags: list) -> int:
        """메타데이터 품질 (25점 만점)"""
        s = 0
        if title and len(title) <= 100: s += 5         # 제목 존재+길이
        if '#' in description: s += 5                   # 해시태그 포함
        if len(tags) >= 5: s += 5                       # 태그 5개 이상
        if any(k in description for k in ['구독', '좋아요', '🔔']): s += 5  # CTA 포함
        if '⚠️' in description or '주의' in description: s += 5  # 면책조항
        self.scores['metadata'] = min(s, 25)
        return self.scores['metadata']

    def total_score(self) -> int:
        return sum(self.scores.values())

    def is_pass(self) -> bool:
        return self.total_score() >= self.PASS_THRESHOLD

    def report(self) -> dict:
        return {
            'scores': self.scores,
            'total': self.total_score(),
            'threshold': self.PASS_THRESHOLD,
            'pass': self.is_pass()
        }

    def evaluate_all(self, script_json: dict, video_path: str, title: str, description: str, tags: list) -> dict:
        """전체 4항목 평가 원스톱"""
        self.score_script(script_json)
        self.score_video(video_path)
        self.score_audio(script_json, video_path)
        self.score_metadata(title, description, tags)
        return self.report()
```

## 3. scripts/run_qa_check.py 생성 (CLI)

파일 경로: `/data/shortflow/scripts/run_qa_check.py`

```python
#!/usr/bin/env python3
"""QA 점수 확인 CLI
사용: python3 scripts/run_qa_check.py --script output/economy_script.json --video output/economy.mp4
"""
import argparse, json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
from engine.qa_score_engine import QAScoreEngine

def main():
    p = argparse.ArgumentParser()
    p.add_argument('--script', required=True, help='대본 JSON 파일 경로')
    p.add_argument('--video', default=None, help='영상 MP4 경로 (없으면 영상 점수 skip)')
    p.add_argument('--title', default='테스트 제목')
    p.add_argument('--description', default='')
    p.add_argument('--tags', default='경제,재테크,shorts')
    a = p.parse_args()

    with open(a.script, 'r', encoding='utf-8') as f:
        script = json.load(f)

    qa = QAScoreEngine()
    qa.score_script(script)
    if a.video and Path(a.video).exists():
        qa.score_video(a.video)
    else:
        print("[SKIP] 영상 파일 없음, video score = 0")
    qa.score_audio(script)
    qa.score_metadata(a.title, a.description, a.tags.split(','))

    r = qa.report()
    print(f"\n{'='*40}")
    for k, v in r['scores'].items():
        print(f"  {k:12}: {v}/25")
    print(f"{'='*40}")
    print(f"  총점: {r['total']}/100")
    print(f"  기준: {r['threshold']}점")
    print(f"  결과: {'✅ PASS' if r['pass'] else '❌ FAIL'}")

if __name__ == '__main__': main()
```
