```typescript
const WORKER_BASE = "http://127.0.0.1:8000";

// ── GET /api/accounts ─────────────────────────────────────────────
export async function GET(req: NextRequest) {
  const { searchParams } = new URL(req.url);
  const authorization = req.headers.get("authorization") ?? "";

  // 1) Try to proxy to FastAPI worker
  try {
    const params = new URLSearchParams();
    if (searchParams.get("platform")) params.set("platform", searchParams.get("platform")!);
    if (searchParams.get("is_active")) params.set("is_active", searchParams.get("is_active")!);

    const workerUrl = `${WORKER_BASE}/api/accounts${params.size ? "?" + params : ""}`;
    const workerRes = await fetch(workerUrl, {
      headers: {
        Authorization: authorization,
        "Content-Type": "application/json",
      },
      signal: AbortSignal.timeout(5000),
    });

    if (workerRes.ok) {
      const data = await workerRes.json();
      return NextResponse.json(data);
    }
  } catch {
    // Worker unavailable — fall through to mock
  }

  // 2) Mock fallback: return accounts.json data (filtered)
  let accounts = loadMockAccounts();
  const platform = searchParams.get("platform");
  const isActive = searchParams.get("is_active");
  if (platform) accounts = accounts.filter((a) => a.platform === platform);
  if (isActive === "true") accounts = accounts.filter((a) => a.is_active === true);
  if (isActive === "false") accounts = accounts.filter((a) => a.is_active === false);

  return NextResponse.json({ accounts, source: "mock" });
}

// ── POST /api/accounts ────────────────────────────────────────────
export async function POST(req: NextRequest) {
  const authorization = req.headers.get("authorization") ?? "";
  const body = await req.json();

  try {
    const workerRes = await fetch(`${WORKER_BASE}/api/accounts`, {
      method: "POST",
      headers: {
        Authorization: authorization,
        "Content-Type": "application/json",
      },
      body: JSON.stringify(body),
      signal: AbortSignal.timeout(5000),
    });
    const data = await workerRes.json();
    return NextResponse.json(data, { status: workerRes.status });
  } catch {
    return NextResponse.json(
      { success: false, detail: "Worker 서버에 연결할 수 없습니다. 나중에 다시 시도해주세요." },
      { status: 503 }
    );
  }
}
```

#### `/data/shortflow/saas-dashboard/app/api/accounts/[id]/route.ts`
```typescript
/**
 * SF-T033: /api/accounts/[id] — Next.js Route Handler
 *
 * PUT    /api/accounts/[id]               — 수정
 * DELETE /api/accounts/[id]               — 연결 해제 (soft delete)
 */

import { NextRequest, NextResponse } from "next/server";

const WORKER_BASE = "http://127.0.0.1:8000";

async function proxyToWorker(
  req: NextRequest,
  method: string,
  id: string,
  body?: unknown
): Promise<NextResponse> {
  const authorization = req.headers.get("authorization") ?? "";
  try {
    const init: RequestInit = {
      method,
      headers: {
        Authorization: authorization,
        "Content-Type": "application/json",
      },
      signal: AbortSignal.timeout(5000),
    };
    if (body !== undefined) init.body = JSON.stringify(body);

    const workerRes = await fetch(`${WORKER_BASE}/api/accounts/${id}`, init);
    const data = await workerRes.json();
    return NextResponse.json(data, { status: workerRes.status });
  } catch {
    return NextResponse.json(
      { success: false, detail: "Worker 서버에 연결할 수 없습니다." },
      { status: 503 }
    );
  }
}

export async function PUT(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  const body = await req.json();
  return proxyToWorker(req, "PUT", id, body);
}

export async function DELETE(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  return proxyToWorker(req, "DELETE", id);
}
```

```typescript
/**
 * SF-T033: /api/accounts/[id]/refresh-token — Next.js Route Handler
 *
 * POST /api/accounts/[id]/refresh-token — 토큰 갱신 (worker 프록시)
 */

import { NextRequest, NextResponse } from "next/server";

const WORKER_BASE = "http://127.0.0.1:8000";

export async function POST(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  const authorization = req.headers.get("authorization") ?? "";
  try {
    const workerRes = await fetch(
      `${WORKER_BASE}/api/accounts/${id}/refresh-token`,
      {
        method: "POST",
        headers: {
          Authorization: authorization,
          "Content-Type": "application/json",
        },
        signal: AbortSignal.timeout(10000),
      }
    );
    const data = await workerRes.json();
    return NextResponse.json(data, { status: workerRes.status });
  } catch {
    return NextResponse.json(
      { success: false, message: "Worker 서버에 연결할 수 없습니다." },
      { status: 503 }
    );
  }
}
```

2. HTTPS push — `.env`에 GITHUB_TOKEN(PAT) 없음

**조치 필요**: CEO가 GitHub PAT 발급 후 `.env`에 `GITHUB_TOKEN=ghp_xxx` 등록
