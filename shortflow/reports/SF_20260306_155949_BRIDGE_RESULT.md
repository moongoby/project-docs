```python
# ── 채널별 메타데이터 프리셋 로드 ──────────────────────────────────────────
_METADATA_PATH = ROOT / "config" / "upload_metadata.json"
try:
    with open(str(_METADATA_PATH), encoding="utf-8") as _f:
        UPLOAD_METADATA = json.load(_f)
except Exception:
    UPLOAD_METADATA = {}


def _build_description(channel: str, title: str, summary: str = "", cta: str = "") -> str:
    """채널 프리셋 description_template 적용."""
    preset = UPLOAD_METADATA.get(channel, {})
    template = preset.get("description_template", "{title}\n\n{hashtags}")
    hashtags = " ".join(preset.get("hashtags", []))
    return template.format(
        title=title,
        summary=summary or "",
        cta=cta or "",
        hashtags=hashtags,
    )


def _get_tags(channel: str):
    preset = UPLOAD_METADATA.get(channel, {})
    return preset.get("tags", CHANNEL_TAGS.get(channel, ["Shorts"]))
# ──────────────────────────────────────────────────────────────────────────
```
