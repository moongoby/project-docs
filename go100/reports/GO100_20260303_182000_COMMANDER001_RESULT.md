        output_schema: dict | None = None,
    ) -> None:
        # agent_name, system_prompt, available_tools, output_schema 초기화

    async def run(self, context: dict) -> dict:
        # 타임아웃 10초, 재시도 2회
        # 파싱 실패 시 {"error": reason} 반환
        for attempt in range(1, _MAX_RETRIES + 1):
            try:
                raw_text = await asyncio.wait_for(
                    self._call_gemini(prompt),
                    timeout=_TIMEOUT_SECONDS,  # 10초
                )
                result = self._parse_json(raw_text)
                return result
            except (asyncio.TimeoutError, json.JSONDecodeError, Exception) as e:
                last_error = f"{type(e).__name__}: {e} (attempt {attempt})"
                ...
        return {"error": last_error, "agent_name": self.agent_name}

### 마이그레이션 SQL

```sql
-- ── 1. 에이전트 분석 보고서 ────────────────────────────────────────────────
CREATE TABLE IF NOT EXISTS go100_agent_reports (
    id          BIGSERIAL PRIMARY KEY,
    report_date DATE          NOT NULL,
    agent_name  VARCHAR(50)   NOT NULL,
    stock_code  VARCHAR(20),
    report_json JSONB         NOT NULL,
    conviction  INTEGER       CHECK (conviction BETWEEN 0 AND 100),
    signal      VARCHAR(20),
    created_at  TIMESTAMPTZ   NOT NULL DEFAULT NOW()
);

-- ── 2. Bull vs Bear 토론 로그 ─────────────────────────────────────────────
CREATE TABLE IF NOT EXISTS go100_debate_log (
    id                  BIGSERIAL PRIMARY KEY,
    debate_date         DATE          NOT NULL,
    stock_code          VARCHAR(20)   NOT NULL,
    round_num           INTEGER       NOT NULL DEFAULT 1,
    bull_argument       TEXT,
    bear_argument       TEXT,
    winner              VARCHAR(10)   CHECK (winner IN ('bull', 'bear', 'draw', NULL)),
    commander_decision  VARCHAR(20),
    created_at          TIMESTAMPTZ   NOT NULL DEFAULT NOW()
);

-- ── 3. 에이전트 성과 추적 ─────────────────────────────────────────────────
CREATE TABLE IF NOT EXISTS go100_agent_performance (
    id                 BIGSERIAL PRIMARY KEY,
    eval_date          DATE          NOT NULL,
    agent_name         VARCHAR(50)   NOT NULL,
    total_signals      INTEGER       NOT NULL DEFAULT 0,
    correct_signals    INTEGER       NOT NULL DEFAULT 0,
    accuracy           NUMERIC(5,4)  CHECK (accuracy BETWEEN 0 AND 1),
    contribution_score NUMERIC(8,4),
    weight_adjustment  NUMERIC(5,4),
    created_at         TIMESTAMPTZ   NOT NULL DEFAULT NOW()
);
```
