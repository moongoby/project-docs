## 최종 수정본 (scripts/go100/generate_closing_report.py)

```python
#!/usr/bin/env python3
"""GO100 장마감 리포트 자동 생성 (T-011)
매일 15:35 KST 크론 실행. 당일 모의투자/실매매 실적 요약."""
import asyncio, json, os, sys
from datetime import date, datetime

sys.path.insert(0, "/root/kis-autotrade-v4")

async def main():
    from backend.app.core.database import AsyncSessionLocal
    from sqlalchemy import text
    today = date.today().isoformat()
    report_lines = [f"# GO100 Closing Report — {today}\n"]

    async with AsyncSessionLocal() as db:
        # 1. 모의투자 오늘 거래
        r = await db.execute(text(
            f"SELECT count(*) FROM go100_paper_trades WHERE executed_at::date = '{today}'"))
        paper_cnt = r.scalar() or 0
        report_lines.append(f"## 모의투자: 오늘 {paper_cnt}건\n")

        # 2. 실매매 오늘 주문
        r2 = await db.execute(text(
            f"SELECT count(*) FROM go100_live_orders WHERE created_at::date = '{today}'"))
        live_cnt = r2.scalar() or 0
        report_lines.append(f"## 실매매 주문: 오늘 {live_cnt}건\n")

        # 3. 리스크 상태
        r3 = await db.execute(text(
            "SELECT count(*) FROM go100_risk_events WHERE created_at::date = CURRENT_DATE"))
        risk_cnt = r3.scalar() or 0
        report_lines.append(f"## 리스크 이벤트: {risk_cnt}건\n")

        # 4. 활성 세션
        r4 = await db.execute(text(
            "SELECT count(*) FROM go100_paper_trading_sessions WHERE status='ACTIVE'"))
        active_cnt = r4.scalar() or 0
        report_lines.append(f"## 활성 모의투자 세션: {active_cnt}개\n")

    report_text = "\n".join(report_lines)
    report_dir = "/root/project-docs/go100/reports"
    fname = f"CUR-GO100-CLOSING-REPORT-001-{today.replace('-','')}.md"
    fpath = os.path.join(report_dir, fname)

    with open(fpath, "w") as f:
        f.write(report_text)
    print(f"CLOSING_REPORT_SAVED={fpath}")

    # git push (root cron 실행이므로 권한 있음)
    os.system(f'cd /root/project-docs && git add -A && git commit -m "[GO100] auto closing report {today}" && git push origin master 2>/dev/null')
    print(f"CLOSING_REPORT_PUSHED={fname}")

if __name__ == "__main__":
    asyncio.run(main())
```
