import asyncio, asyncpg
async def main():
    conn = await asyncpg.connect('postgresql://kis_admin:KisAuto2026!Secure@localhost:5432/kisautotrade')

    sessions = await conn.fetch('''
        SELECT session_id, user_id, strategy_card_id, start_date, end_date, status,
               initial_capital, current_capital, total_trades,
               (SELECT count(*) FROM go100_paper_trades WHERE session_id=s.session_id) as actual_trade_count,
               (SELECT max(executed_at) FROM go100_paper_trades WHERE session_id=s.session_id) as last_trade
        FROM go100_paper_trading_sessions s WHERE status='ACTIVE' ORDER BY session_id
    ''')
    print('=== ACTIVE Sessions ===')
    for s in sessions: print(dict(s))

    trades = await conn.fetch('''
        SELECT trade_id, session_id, ticker, trade_type, quantity, price, executed_at, signal_source, pnl
        FROM go100_paper_trades ORDER BY executed_at DESC LIMIT 10
    ''')
    print('\n=== Recent 10 Trades ===')
    for t in trades: print(dict(t))

    daily = await conn.fetch('''
        SELECT date_trunc('day', executed_at) as day, count(*) as cnt
        FROM go100_paper_trades WHERE executed_at >= CURRENT_DATE - interval '7 days'
        GROUP BY 1 ORDER BY 1
    ''')
    print('\n=== Daily Trade Count (7d) ===')
    for d in daily: print(dict(d))

    all_sessions = await conn.fetch('''
        SELECT status, count(*) as cnt FROM go100_paper_trading_sessions GROUP BY status
    ''')
    print('\n=== All Sessions by Status ===')
    for s in all_sessions: print(dict(s))

    await conn.close()
asyncio.run(main())

### STEP 10: 0거래 원인 심층 분석

#### 원인 A — T-017B 이전 KeyError (이미 수정)
- `indicator_precompute()` 내 pandas 3.0 호환 코드:
  ```python
  out = out.groupby("stock_code", group_keys=True).apply(_add_indicators, include_groups=False)
  out = out.reset_index(level=0).reset_index(drop=True)
  ```
- pandas 3.0에서 `include_groups=False` 시 stock_code 컬럼이 그룹 함수에 전달되지 않음
- `group_keys=True`로 외부 인덱스에 stock_code 유지 → `reset_index(level=0)`로 복원
- 테스트 결과: 현재 코드 정상 동작 확인 (pandas 3.0.1)

#### 원인 B — entry_rules 포맷 불일치 (미해결)
- 전략카드 35 `entry_rules`:
  ```json
  {"logic": "AND", "conditions": [
    {"params": {"fast": 5, "slow": 20}, "indicator": "golden_cross"},
    {"params": {"period": 20, "threshold": 2.0}, "indicator": "volume_surge"}
  ], "entry_type": "market", "position_size": 0.1}
  ```
- `SignalEvaluator._eval_one_entry()` 기대 포맷:
  ```json
  {"type": "ma_cross", "short": 5, "long": 20, "direction": "golden"}
  ```
- 카드에 `"type"` 키 없음 → `t = ""` → 모든 조건 False → 매수 시그널 미발생
- **후속 조치 필요**: 전략카드 entry_rules 수정 또는 SignalEvaluator에 `indicator` 키 파싱 추가
