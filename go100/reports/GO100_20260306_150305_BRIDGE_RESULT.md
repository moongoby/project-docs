  const [agents, setAgents] = useState<MiniAgent[]>([]);
  const [latestDebate, setLatestDebate] = useState<MiniDebate | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    setLoading(true);
    Promise.all([
      commanderApi.getStatus().catch(() => null),
      commanderApi.getDebates(1, 0).catch(() => null),
    ]).then(([st, db]) => {
      if (Array.isArray(st)) setAgents(st.slice(0, 10));
      const debates = Array.isArray(db) ? db : [];
      if (debates.length > 0) setLatestDebate(debates[0]);
    }).finally(() => setLoading(false));
  }, []);
  ...
}
