- 시스템 프롬프트:

  ```
  "너는 강세 관점 투자 분석가다. 수급, 기술적 분석, 뉴스 보고서를 받아서 이 종목을

  ```
- 입력: `{"stock_code": "...", "supply_report": {...}, "technical_report": {...}, "news_report": {...}, "bear_argument": null|"..."}`
- 출력: `{"agent": "bull", "stock_code": "...", "argument": "...", "confidence": 0-100, "key_evidence": ["..."], "rebuttal": null|"..."}`
- `_build_prompt()` 오버라이드: bear_argument 있으면 반박 지시 포함
- `_mock_llm_response()`: LLMGateway 미설치 시 테스트 mock (confidence=72)

### 파일 내용 전문
```python
# DIR-GO100-COMMANDER-004, 2026-03-03
"""
bull_agent.py — GO100 강세(Bull) 관점 토론 에이전트

- BaseAgent 상속
- 수급/기술/뉴스 보고서를 입력받아 매수 논거를 구성
- Bear 반박이 있으면 데이터 기반 재반박
- 반드시 JSON 응답
"""
from __future__ import annotations

import json
import logging
from typing import Any, Dict, List, Optional

from .base_agent import BaseAgent

logger = logging.getLogger(__name__)

BULL_SYSTEM_PROMPT = (
    "너는 강세 관점 투자 분석가다. "
    "수급, 기술적 분석, 뉴스 보고서를 받아서 이 종목을 매수해야 하는 논거를 최대한 강하게 구성한다. "
    "반대 관점(Bear)의 논거가 주어지면 그것을 구체적으로 반박한다. "
    "감정적이지 않고 데이터 기반으로 논증하라. "
    "반드시 JSON으로 응답하라."
)

OUTPUT_SCHEMA: Dict[str, Any] = {
    "agent": "bull",
    "stock_code": "str",
    "argument": "str — 매수 논거 (데이터 기반, 200자 이상)",
    "confidence": "int 0-100",
    "key_evidence": ["str — 핵심 근거 1", "str — 핵심 근거 2", "..."],
    "rebuttal": "null | str — Bear 논거 반박 (bear_argument 있을 때만)",
}


class BullAgent(BaseAgent):
    def __init__(self) -> None:
        super().__init__(
            agent_name="bull",
            system_prompt=BULL_SYSTEM_PROMPT,
            available_tools={},
            output_schema=OUTPUT_SCHEMA,
            timeout=15.0,
            max_retries=2,
        )

    def _build_prompt(self, context: dict, tool_results: dict) -> str:
        stock_code = context.get("stock_code", "?")
        supply_report = context.get("supply_report", {})
        technical_report = context.get("technical_report", {})
        news_report = context.get("news_report", {})
        bear_argument: Optional[str] = context.get("bear_argument")

        parts: List[str] = [
            f"## 종목 코드: {stock_code}",
            "\n### 수급 보고서\n```json\n" + json.dumps(supply_report, ensure_ascii=False, indent=2) + "\n```",
            "\n### 기술적 분석 보고서\n```json\n" + json.dumps(technical_report, ensure_ascii=False, indent=2) + "\n```",
            "\n### 뉴스 보고서\n```json\n" + json.dumps(news_report, ensure_ascii=False, indent=2) + "\n```",
        ]

        if bear_argument:
            parts.append(f"\n### Bear의 논거 (아래를 데이터로 반박하라)\n{bear_argument}")
            parts.append("\n위 Bear 논거의 약점을 구체적인 수급/기술/뉴스 데이터로 반박하고, 강세 관점을 더욱 강화하라.")
        else:
            parts.append("\n위 데이터를 종합하여 이 종목을 지금 매수해야 하는 가장 강력한 논거를 구성하라.")

        parts.append("\n## 출력 형식 (순수 JSON만 반환, 다른 텍스트 없이)\n```json\n" + json.dumps(OUTPUT_SCHEMA, ensure_ascii=False, indent=2) + "\n```")
        return "\n".join(parts)

    def _mock_llm_response(self) -> dict:
        return {
            "agent": "bull", "stock_code": "TEST",
            "argument": "[MOCK-BULL] 외국인 5일 연속 순매수(+42억)로 수급 개선이 확인된다...",
            "confidence": 72,
            "key_evidence": ["외국인 5일 연속 순매수 +42억", "RSI 40→54 반등", "MACD 골든크로스 임박", "실적 상향 뉴스 3건"],
            "rebuttal": None, "_mock": True,
        }
```

**위치:** `/root/kis-autotrade-v4/backend/app/services/go100/agents/debate.py`

### 3라운드 흐름
```
Round 1: Bull 초기 논거 ←→ Bear 초기 논거  (asyncio.gather 병렬)
Round 2: Bull이 Bear 반박 ←→ Bear가 Bull 반박  (병렬)
Round 3: Bull 최종 논거 ←→ Bear 최종 논거  (병렬)
각 라운드 → go100_debate_log INSERT
```

```python
    db=None,  # AsyncSession (None이면 DB 저장 건너뜀)

    ...
    return {
        "winner": "BULL" | "BEAR" | "DRAW",
        "bull_final_confidence": int,
        "bear_final_confidence": int,
        "elapsed_sec": float,
        "rounds": [{"round": 1, "bull": {...}, "bear": {...}}, ...]
    }
```

### winner 판정 로직
```python
DRAW_THRESHOLD = 5   # ±5pt 이내면 DRAW
def _determine_winner(bull_conf: int, bear_conf: int) -> str:
    diff = bull_conf - bear_conf
    if abs(diff) <= DRAW_THRESHOLD:
        return "DRAW"
    return "BULL" if diff > 0 else "BEAR"
```

- Round 1/2: winner=NULL
- Round 3: winner = 최종 판정 (bull/bear/draw) 소문자
