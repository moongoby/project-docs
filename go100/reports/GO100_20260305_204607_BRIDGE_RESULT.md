```python
"""
GO100 에러 모니터링 미들웨어 (T-031)
500 에러 발생 시 DB 기록 + Telegram 알림
"""
import traceback
import logging
import os
from typing import Callable

from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.responses import JSONResponse

logger = logging.getLogger("error_monitor")


class ErrorMonitorMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next: Callable) -> Response:
        try:
            response = await call_next(request)
            if response.status_code >= 500:
                await self._handle_error(
                    request=request,
                    status_code=response.status_code,
                    error_message=f"HTTP {response.status_code}",
                    tb="",
                )
            return response
        except Exception as exc:
            tb = traceback.format_exc()
            await self._handle_error(
                request=request,
                status_code=500,
                error_message=str(exc),
                tb=tb,
            )
            return JSONResponse(
                status_code=500,
                content={"detail": "Internal Server Error"},
            )

    async def _handle_error(self, request, status_code, error_message, tb):
        endpoint = str(request.url.path)
        method = request.method
        request_body = {}
        try:
            body_bytes = await request.body()
            if body_bytes:
                import json
                try:
                    request_body = json.loads(body_bytes[:4096])
                except Exception:
                    request_body = {"raw": body_bytes[:500].decode("utf-8", errors="replace")}
        except Exception:
            pass

        logger.error("[ErrorMonitor] %s %s → %s | %s", method, endpoint, status_code, error_message)
        if tb:
            logger.error("[ErrorMonitor] Traceback:\n%s", tb)

        await self._save_to_db(endpoint, status_code, error_message, tb, request_body)
        await self._send_telegram(method, endpoint, status_code, error_message, tb)

    async def _save_to_db(self, endpoint, status_code, error_message, tb, request_body):
        try:
            from backend.app.core.database import AsyncSessionLocal
            from sqlalchemy import text
            import json
            async with AsyncSessionLocal() as session:
                await session.execute(
                    text("""
                        INSERT INTO go100_error_log
                            (endpoint, status_code, error_message, traceback, request_body)
                        VALUES
                            (:endpoint, :status_code, :error_message, :traceback, CAST(:request_body AS jsonb))
                    """),
                    {
                        "endpoint": endpoint[:500],
                        "status_code": status_code,
                        "error_message": error_message[:2000],
                        "traceback": tb[:8000],
                        "request_body": json.dumps(request_body, ensure_ascii=False),
                    },
                )
                await session.commit()
        except Exception as e:
            logger.error("[ErrorMonitor] DB 저장 실패: %s", e)

    async def _send_telegram(self, method, endpoint, status_code, error_message, tb):
        try:
            bot_token = os.getenv("GO100_TELEGRAM_BOT_TOKEN", "")
            chat_id = os.getenv("GO100_TELEGRAM_CHAT_ID", "")
            if not (bot_token and chat_id):
                logger.info("[ErrorMonitor] Telegram 미설정 — 알림 생략")
                return
            tb_summary = tb[-800:] if tb else ""
            message = (
                f"🚨 <b>[GO100 500 에러]</b>\n"
                f"<b>Endpoint:</b> {method} {endpoint}\n"
                f"<b>Status:</b> {status_code}\n"
                f"<b>Error:</b> {error_message[:300]}\n"
                + (f"<b>Traceback:</b>\n<pre>{tb_summary}</pre>" if tb_summary else "")
            )
            import httpx
            url = f"https://api.telegram.org/bot{bot_token}/sendMessage"
            async with httpx.AsyncClient() as client:
                resp = await client.post(
                    url,
                    json={"chat_id": chat_id, "text": message[:4096], "parse_mode": "HTML"},
                    timeout=10,
                )
                if resp.status_code != 200:
                    logger.error("[ErrorMonitor] Telegram 발송 실패: %s", resp.status_code)
        except Exception as e:
            logger.error("[ErrorMonitor] Telegram 알림 실패: %s", e)
```

### STEP 5: DB 마이그레이션 파일 생성

파일: `backend/migrations/065_add_error_log_table.sql`
(지시서에서 050_을 지정했으나 050_*가 이미 3개 존재하여 충돌 방지를 위해 065_ 사용)

```sql
CREATE TABLE IF NOT EXISTS go100_error_log (
    id           SERIAL PRIMARY KEY,
    endpoint     TEXT,
    status_code  INT,
    error_message TEXT,
    traceback    TEXT,
    request_body JSONB,
    created_at   TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_go100_error_log_created_at ON go100_error_log (created_at DESC);
CREATE INDEX IF NOT EXISTS idx_go100_error_log_status_code ON go100_error_log (status_code);
```
