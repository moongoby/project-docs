'use client';
import { useEffect, useState } from 'react';
import { go100Api } from '@/lib/go100/api/go100Api';
export default function GO100Dashboard() {
  const [overview, setOverview] = useState<any>(null);
  const [risk, setRisk] = useState<any>(null);
  const [loading, setLoading] = useState(true);
  useEffect(() => {
    Promise.all([go100Api.getOverview(), go100Api.getRiskStatus()])
      .then(([o, r]) => { setOverview(o); setRisk(r); })
      .catch(console.error)
      .finally(() => setLoading(false));
  }, []);
  if (loading) return <div className="flex items-center justify-center h-64"><div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-500" /></div>;
  return (
    <div className="p-4 md:p-6 space-y-6">
      <h1 className="text-2xl font-bold dark:text-white">GO100 대시보드</h1>
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
        <div className="bg-white dark:bg-gray-800 rounded-xl p-4 shadow"><p className="text-sm text-gray-500">총 자산</p><p className="text-2xl font-bold">{overview?.total_asset?.toLocaleString() ?? '-'}원</p></div>
        <div className="bg-white dark:bg-gray-800 rounded-xl p-4 shadow"><p className="text-sm text-gray-500">수익률</p><p className="text-2xl font-bold">{overview?.return_pct?.toFixed(2) ?? '-'}%</p></div>
        <div className="bg-white dark:bg-gray-800 rounded-xl p-4 shadow"><p className="text-sm text-gray-500">활성 전략</p><p className="text-2xl font-bold">{overview?.active_strategies ?? '-'}개</p></div>
        <div className="bg-white dark:bg-gray-800 rounded-xl p-4 shadow"><p className="text-sm text-gray-500">리스크</p><p className={`text-2xl font-bold ${risk?.kill_switch ? 'text-red-500' : 'text-green-500'}`}>{risk?.kill_switch ? 'KILL' : 'NORMAL'}</p></div>
      </div>

}
