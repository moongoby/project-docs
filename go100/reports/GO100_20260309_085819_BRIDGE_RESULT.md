    async def verify_trade_facts(self, stock_code, price, action, db=None)

        # LLM 주장 vs 실데이터 수치 비교 (5% 이상 괴리 감지)

    async def paper_trade_first(self, order)
        # GO100_LIVE_TRADING_ENABLED=false → 항상 모의투자 안내

        # go100_paper_trades에서 진입 근거 vs 실제 손익 대조
