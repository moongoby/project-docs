## 4. 스크립트 내용 (generate_manager_snapshot.py)

```python
#!/usr/bin/env python3
"""
GO100 총괄매니저 실시간 스냅샷 생성기 (T-039)
CEO 및 외부 AI(Claude, Genspark)가 인증 없이 서버 상태를 확인할 수 있도록
frontend/public/manager/ 에 정적 JSON 파일을 생성합니다.
"""
import json
import datetime
import subprocess
import pathlib
import sys

import psycopg2
import psycopg2.extras

OUTPUT_DIR = pathlib.Path("/root/kis-autotrade-v4/frontend/public/manager")
OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

DB_CONFIG = {
    "host": "localhost",
    "port": 5432,
    "dbname": "kisautotrade",
    "user": "kis_admin",
    "password": "KisAuto2026!Secure",
}

KST = datetime.timezone(datetime.timedelta(hours=9))

def now_kst() -> str:
    return datetime.datetime.now(KST).isoformat()

def get_db():
    return psycopg2.connect(**DB_CONFIG)

def systemctl_is_active(service: str) -> str:
    try:
        result = subprocess.run(
            ["systemctl", "is-active", service],
            capture_output=True, text=True, timeout=5
        )
        return result.stdout.strip()
    except Exception:
        return "unknown"

def run_query(conn, sql: str, params=None) -> list:
    try:
        with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
            cur.execute(sql, params)
            rows = cur.fetchall()
            return [dict(r) for r in rows]
    except Exception as e:
        print(f"[WARN] query failed: {e}", file=sys.stderr)
        return []

def run_scalar(conn, sql: str, params=None):
    try:
        with conn.cursor() as cur:
            cur.execute(sql, params)
            row = cur.fetchone()
            return row[0] if row else None
    except Exception as e:
        print(f"[WARN] scalar query failed: {e}", file=sys.stderr)
        return None

def table_exists(conn, table_name: str) -> bool:
    result = run_scalar(
        conn,
        "SELECT EXISTS(SELECT 1 FROM information_schema.tables WHERE table_name=%s)",
        (table_name,)
    )
    return bool(result)

def serialize_value(v):
    if isinstance(v, (datetime.datetime, datetime.date)):
        return v.isoformat()
    if hasattr(v, '__float__'):
        try:
            return float(v)
        except Exception:
            return str(v)
    return v

def serialize_row(row: dict) -> dict:
    return {k: serialize_value(v) for k, v in row.items()}

def build_snapshot(conn) -> dict:
    services = ["go100", "go100-frontend", "redis", "postgresql"]
    service_status = {s: systemctl_is_active(s) for s in services}

    total_tables = run_scalar(
        conn,
        "SELECT count(*) FROM information_schema.tables WHERE table_schema='public'"
    ) or 0
    go100_tables = run_scalar(
        conn,
        "SELECT count(*) FROM information_schema.tables WHERE table_schema='public' AND table_name LIKE 'go100_%'"
    ) or 0
    agent_tools = run_scalar(conn, "SELECT count(*) FROM go100_agent_performance") or 0 \
        if table_exists(conn, "go100_agent_performance") else 0

    db_summary = {
        "total_tables": int(total_tables),
        "go100_tables": int(go100_tables),
        "agent_tool_records": int(agent_tools),
    }

    paper_trading = []
    if table_exists(conn, "go100_paper_trading_sessions"):
        rows = run_query(
            conn,
            """SELECT session_id, status, total_trades, initial_capital, current_capital,
                      start_date, end_date
               FROM go100_paper_trading_sessions
               ORDER BY created_at DESC LIMIT 5"""
        )
        paper_trading = [serialize_row(r) for r in rows]

    v3_model = {"active": False, "model_version": None, "last_prediction": None}
    if table_exists(conn, "go100_strategy_cards"):
        row = run_query(conn, """SELECT 1 FROM go100_strategy_cards LIMIT 1""")
        v3_model["active"] = len(row) > 0
    if table_exists(conn, "go100_agent_reports"):
        last_pred = run_scalar(conn, "SELECT MAX(created_at) FROM go100_agent_reports")
        v3_model["last_prediction"] = serialize_value(last_pred)

    commander = {"agent_count": 0, "avg_accuracy": None, "recent_debates": 0}
    if table_exists(conn, "go100_agent_performance"):
        agent_count = run_scalar(
            conn,
            "SELECT count(DISTINCT agent_name) FROM go100_agent_performance"
        ) or 0
        avg_acc = run_scalar(
            conn,
            "SELECT round(avg(accuracy)::numeric, 4) FROM go100_agent_performance WHERE accuracy IS NOT NULL"
        )
        commander["agent_count"] = int(agent_count)
        commander["avg_accuracy"] = float(avg_acc) if avg_acc is not None else None

    if table_exists(conn, "go100_debate_log"):
        recent_debates = run_scalar(
            conn,
            "SELECT count(*) FROM go100_debate_log WHERE debate_date >= current_date - interval '7 days'"
        ) or 0
        commander["recent_debates"] = int(recent_debates)

    cron_status: list = []
    try:
        result = subprocess.run(
            ["journalctl", "-u", "cron", "--no-pager", "-n", "20", "--output=short"],
            capture_output=True, text=True, timeout=5
        )
        lines = [l for l in result.stdout.splitlines() if "go100" in l.lower() or "manager" in l.lower()]
        cron_status = lines[-5:]
    except Exception:
        cron_status = []

    return {
        "service_status": service_status,
        "db_summary": db_summary,
        "paper_trading": paper_trading,
        "v3_model": v3_model,
        "commander": commander,
        "cron_status": cron_status,
        "updated_at": now_kst(),
    }

def build_trades(conn) -> list:
    if not table_exists(conn, "go100_paper_trades"):
        return []
    rows = run_query(
        conn,
        """SELECT ticker AS symbol, trade_type AS side, quantity, price, executed_at, pnl
           FROM go100_paper_trades
           ORDER BY executed_at DESC LIMIT 50"""
    )
    return [serialize_row(r) for r in rows]

def build_agents(conn) -> dict:
    performance = []
    if table_exists(conn, "go100_agent_performance"):
        rows = run_query(
            conn,
            """SELECT eval_date, agent_name, total_signals, correct_signals,
                      accuracy, contribution_score, weight_adjustment
               FROM go100_agent_performance
               ORDER BY eval_date DESC, agent_name"""
        )
        performance = [serialize_row(r) for r in rows]

    debates = []
    if table_exists(conn, "go100_debate_log"):
        rows = run_query(
            conn,
            """SELECT debate_date, stock_code, round_num, winner, commander_decision,
                      bull_confidence, bear_confidence, created_at
               FROM go100_debate_log
               ORDER BY created_at DESC LIMIT 20"""
        )
        debates = [serialize_row(r) for r in rows]

    return {
        "performance": performance,
        "recent_debates": debates,
        "updated_at": now_kst(),
    }

def build_errors(conn) -> list:
    if not table_exists(conn, "go100_error_log"):
        return []
    rows = run_query(
        conn,
        """SELECT endpoint, status_code, error_message, created_at
           FROM go100_error_log
           ORDER BY created_at DESC LIMIT 30"""
    )
    return [serialize_row(r) for r in rows]

def main():
    print(f"[{now_kst()}] GO100 Manager Snapshot 생성 시작")
    conn = get_db()
    try:
        snapshot = build_snapshot(conn)
        trades = build_trades(conn)
        agents = build_agents(conn)
        errors = build_errors(conn)
    finally:
        conn.close()

    (OUTPUT_DIR / "snapshot.json").write_text(json.dumps(snapshot, ensure_ascii=False, indent=2), encoding="utf-8")
    (OUTPUT_DIR / "trades.json").write_text(json.dumps(trades, ensure_ascii=False, indent=2), encoding="utf-8")
    (OUTPUT_DIR / "agents.json").write_text(json.dumps(agents, ensure_ascii=False, indent=2), encoding="utf-8")
    (OUTPUT_DIR / "errors.json").write_text(json.dumps(errors, ensure_ascii=False, indent=2), encoding="utf-8")
    (OUTPUT_DIR / "_updated_at.txt").write_text(now_kst(), encoding="utf-8")

    print(f"[{now_kst()}] 완료: {OUTPUT_DIR}")
    print(f"  snapshot.json: {(OUTPUT_DIR / 'snapshot.json').stat().st_size} bytes")
    print(f"  trades.json:   {(OUTPUT_DIR / 'trades.json').stat().st_size} bytes")
    print(f"  agents.json:   {(OUTPUT_DIR / 'agents.json').stat().st_size} bytes")
    print(f"  errors.json:   {(OUTPUT_DIR / 'errors.json').stat().st_size} bytes")

if __name__ == "__main__":
    main()
```
