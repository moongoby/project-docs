class Trigger(Enum): T1~T9  # 거래량폭발/상한가/갭/급등/강세/섹터/체결강도/종가/RSI반전

class CellGrade(Enum):
    OPTIMAL=20pt  # ◎ 최적
    SUITABLE=15pt # ○ 적합
    CONDITIONAL=8pt # △ 조건부
    FORBIDDEN=0pt   # ✕ 금지

class TriggerTacticMatrix:
    def lookup(trigger, tactic) -> MatrixCell
    def get_forbidden_pairs() -> List[Tuple]   # 금지 18셀+
    def get_synergy_pairs() -> List[Tuple]     # OPTIMAL 셀 목록

        # D6→(T2,A2) | D7→(T8,B4),(T4,B4) | S1→(T1,B2)
    def get_match_quality_score(trigger, tactic) -> int
    def matrix_summary() -> Dict[str, int]

### 2-B. ConvictionScoreEngine (`conviction_score.py`)

```python
class ConvictionScoreEngine:
    # 요소 1: DAILY_GRADE (0~25점)
    def calc_daily_grade(dcs_grade: str) -> int
        # A=25, B=20, C=15, D=10, F=0

    # 요소 2: MATCH_QUALITY (0~20점)
    def calc_match_quality(trigger_id, tactic_id, matrix=None) -> int
        # TriggerTacticMatrix 연동

    # 요소 3: TIME_FIT (0~20점)
    def calc_time_fit(strategy_id, current_time: str) -> int
        # PRIMARY=20, SECONDARY=14, NEUTRAL=8, OFF=0
        # D2: 09:00-11:00 PRIMARY | D7: 14:30-15:20 PRIMARY | D6: 15:00-15:30 PRIMARY

    # 요소 4: TECH_CONFIRM (0~20점)
    def calc_tech_confirm(signal_id, pf_ranking: int) -> int
        # TOP3=20, TOP5=16, TOP10=12, LOWER=8, NONE=0

    # 요소 5: MARKET_ENV (0~15점)
    def calc_market_env(regime, volatility) -> int
        # BULL+LOW=15, BULL+MID=12, BULL+HIGH=9
        # FLAT+LOW=10, FLAT+MID=8,  FLAT+HIGH=5
        # BEAR+LOW=5,  BEAR+MID=3,  BEAR+HIGH=0

    def calculate(context: Dict) -> CSResult   # 5요소 합산 (만점 100)
    def judge(cs_score: int) -> CSJudgment
        # ≥65: FULL(1.0x) | 50~64: REDUCED(0.5x) | <50: BLOCKED(0x)
```

### 2-C. ExecutionQualityScoreEngine (`execution_quality_score.py`)

```python
class ExecutionQualityScoreEngine:
    # 요소 1: SLIPPAGE_EST (0~20점)
    def calc_slippage_est(spread_pct, depth=0, surge_pct=0) -> int
        # TIGHT(≤0.2%)=18, NORMAL(≤0.5%)=14, WIDE(≤1.0%)=9, VERY_WIDE(>1%)=4

    # 요소 2: FRESHNESS (0~25점)
    def calc_freshness(bars_since_signal: int) -> int
        # 0BAR=25, 1BAR=20, 2BAR=15, 3BAR=10, 4BAR+=5

    # 요소 3: VOL_QUALITY (0~20점)
    def calc_vol_quality(current_vol, ma20_vol) -> int
        # EXCELLENT(≥200%)=17~20, GOOD(≥150%)=14, NORMAL(≥80%)=9~12, THIN(<80%)=0~8

    # 요소 4: PRICE_POSITION (0~20점)
    def calc_price_position(price, day_range, strategy_type) -> int
        # PULLBACK: 저점(0~30%)=20, 중점=15, 고점=5
        # BREAKOUT: 고점(70%+)=20, 중점=15, 저점=5

    # 요소 5: ORDERBOOK_BALANCE (0~15점)
    def calc_orderbook_balance(bid_vol, ask_vol, vp_ratio, use_proxy) -> int
        # 실거래: bid/ask 비율 | 백테스트: (VP_ratio-1.0)×15 클리핑 0~15
        # ※ 백테스트 VP 프록시 — 실거래 시 실제 호가로 대체 필요

    def calculate(context: Dict) -> EQSResult   # 5요소 합산 (만점 100)
    def judge(eqs_score: int) -> EQSGate
        # ≥65: PROCEED(0% 패널티) | 50~64: REDUCE(+0.05%) |
        # 35~49: LIMIT_ONLY(+0.10%) | <35: REJECT
```

### 괴리 원인 분석

**CS 평균 괴리 (66.7 vs 기대 78.1)**:
1. **DCS 등급 역산 불일치**: 실제 DCS 등급 없어 PnL 분포로 추정 (PnL>2%→A, 등)
   → 실제 DCS는 전날 조건 기반이므로 PnL과 직접 대응 불가
2. **MATCH_QUALITY**: 전략명→트리거/전술 매핑이 #2 보고서 기준과 다를 가능성
3. **결론**: 역산 프록시의 구조적 한계. 실거래 DB에 trigger_id/tactic_id 컬럼 추가 시 정밀 재현 가능

**EQS 평균 ✅ (67.7 vs 기대 65.6)**:
- entry_spread_pct 실데이터 사용 → SLIPPAGE_EST 정확
- FRESHNESS, PRICE_POSITION 추정값 사용이지만 분포 유사
- VP_ratio 기반 프록시 허용 오차 범위 내

**CS≥65 PF 이상 (11.37 vs 기대 1.55)**:
- 샘플링 RANDOM 변동성 + DCS 역산 편향으로 CS 분포 집중도 차이
- CS≥65 필터 후 남은 샘플의 PnL 분포가 원본과 상이
