```typescript
import { fetchApi } from "./api";
import type {
  AdminUser,
  AdminUserListParams,
  AdminUserListResponse,
  RoleType,
} from "@/types/admin-user";

/** GET /api/users — 관리자 사용자 목록 */
export async function getAdminUsers(
  params?: AdminUserListParams
): Promise<AdminUserListResponse> {
  const search = new URLSearchParams();
  if (params?.page != null) search.set("page", String(params.page));
  if (params?.per_page != null) search.set("per_page", String(params.per_page));
  if (params?.search) search.set("search", params.search);
  if (params?.role) search.set("role", params.role);
  if (params?.status) search.set("status", params.status);
  const qs = search.toString();
  const res = await fetchApi<Partial<AdminUserListResponse>>(
    `users${qs ? `?${qs}` : ""}`
  );
  return {
    data: res?.data ?? [],
    current_page: res?.current_page ?? 1,
    last_page: res?.last_page ?? 1,
    per_page: res?.per_page ?? 20,
    total: res?.total ?? 0,
  };
}

/** GET /api/users/{id} — 관리자 사용자 상세 */
export async function getAdminUser(id: number): Promise<AdminUser> {
  const res = await fetchApi<{ user: AdminUser }>(`users/${id}`);
  return res.user;
}

/** PUT /api/users/{id}/role — 사용자 역할 변경 */
export async function updateAdminUserRole(
  id: number,
  role: RoleType
): Promise<AdminUser> {
  const res = await fetchApi<{ user: AdminUser }>(`users/${id}/role`, {
    method: "PUT",
    body: JSON.stringify({ role }),
  });
  return res.user;
}

/** PUT /api/users/{id}/status — 사용자 상태 토글 (active ↔ inactive) */
export async function toggleAdminUserStatus(id: number): Promise<AdminUser> {
  const res = await fetchApi<{ user: AdminUser }>(`users/${id}/status`, {
    method: "PUT",
  });
  return res.user;
}
```

**결과**: ✅ 완료. 4 함수 구현: getAdminUsers, getAdminUser, updateAdminUserRole, toggleAdminUserStatus.

  { role: "md", label: "MD", color: "bg-purple-100 text-purple-700" },

  { role: "outsource", label: "외주", color: "bg-zinc-100 text-zinc-700" },
];

function countByRole(users: AdminUser[], role: RoleType): number {
  return users.filter((u) => u.roles.includes(role)).length;
}

export default function AdminUsersPage() {
  const [result, setResult] = useState<AdminUserListResponse | null>(null);
  const [loading, setLoading] = useState(true);
  const [search, setSearch] = useState("");
  const [role, setRole] = useState<RoleType | "">("");
  const [status, setStatus] = useState<AdminUserStatus | "">("");
  const [page, setPage] = useState(1);

  const load = useCallback(async () => {
    setLoading(true);
    try {
      const res = await getAdminUsers({
        page,
        per_page: 20,
        ...(search ? { search } : {}),
        ...(role ? { role } : {}),
        ...(status ? { status } : {}),
      });
      setResult(res);
    } catch {
      setResult({ data: [], current_page: 1, last_page: 1, per_page: 20, total: 0 });
    } finally {
      setLoading(false);
    }
  }, [page, search, role, status]);

  useEffect(() => {
    load();
  }, [load]);

  const users = result?.data ?? [];
  const lastPage = result?.last_page ?? 1;
  const currentPage = result?.current_page ?? 1;
  const total = result?.total ?? 0;

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <h1 className="text-xl font-semibold text-zinc-900">사용자 관리</h1>

      {/* 역할별 통계 카드 */}
      <div className="grid grid-cols-3 gap-3 sm:grid-cols-6">
        {ROLE_STAT_LABELS.map(({ role: r, label, color }) => (
          <div
            key={r}
            className="rounded-lg border border-zinc-200 bg-white p-3 text-center"

            <p className="text-xs text-zinc-500">{label}</p>
            <p className={`mt-1 text-lg font-semibold ${color.split(" ")[1]}`}>
              {countByRole(users, r)}
            </p>
          </div>
        ))}
      </div>

      <AdminUserFilter
        search={search}
        role={role}
        status={status}
        onSearchChange={(v) => { setSearch(v); setPage(1); }}
        onRoleChange={(v) => { setRole(v); setPage(1); }}
        onStatusChange={(v) => { setStatus(v); setPage(1); }}
      />

      {loading ? (
        <p className="text-zinc-500">불러오는 중...</p>
      ) : (
        <AdminUserTable users={users} />
      )}

      {lastPage > 1 && (
        <div className="flex justify-center gap-2">
          <button
            type="button"
            disabled={currentPage <= 1}
            onClick={() => setPage((p) => Math.max(1, p - 1))}
            className="rounded-md border border-zinc-200 bg-white px-3 py-1.5 text-sm disabled:opacity-50"
          >
            이전
          </button>
          <span className="flex items-center px-2 text-sm text-zinc-600">
            {currentPage} / {lastPage}
          </span>
          <button
            type="button"
            disabled={currentPage >= lastPage}
            onClick={() => setPage((p) => p + 1)}
            className="rounded-md border border-zinc-200 bg-white px-3 py-1.5 text-sm disabled:opacity-50"
          >
            다음
          </button>
        </div>
      )}
    </div>
  );
}

];

export default function AdminUserDetailPage() {
  const params = useParams();
  const id = Number(params?.id);
  const [user, setUser] = useState<AdminUser | null>(null);
  const [loading, setLoading] = useState(true);
  const [actionLoading, setActionLoading] = useState(false);
  const [selectedRole, setSelectedRole] = useState<RoleType>("retail");
  const [error, setError] = useState<string | null>(null);
  const [success, setSuccess] = useState<string | null>(null);

    if (!id || Number.isNaN(id)) return;
    setLoading(true);
    try {
      const u = await getAdminUser(id);
      setUser(u);
      if (u.roles.length > 0) setSelectedRole(u.roles[0]);

      setUser(null);
    } finally {
      setLoading(false);
    }

    load();

    setError(null);
    setSuccess(null);
    try {
      const updated = await updateAdminUserRole(user.id, selectedRole);
      setUser(updated);

      setActionLoading(false);

      const updated = await toggleAdminUserStatus(user.id);

      setSuccess(`사용자가 ${action}되었습니다.`);
    } catch (e: unknown) {
      setError(e instanceof Error ? e.message : "처리 중 오류가 발생했습니다.");
    } finally {
      setActionLoading(false);
    }

    return <div className="p-6"><p className="text-zinc-500">불러오는 중...</p></div>;

                  }`}

                  {user.status === "active" ? "활성" : "비활성"}
                </span>

                {new Date(user.created_at).toLocaleDateString("ko-KR")}

              </td>

          ))}
        </tbody>
      </table>
    </div>
  );
}
