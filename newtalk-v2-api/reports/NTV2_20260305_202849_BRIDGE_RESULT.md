import { fetchApi } from "./api";
import type {
  DropshipProduct,
  DropshipApplication,
  DropshipOrder,
  DropshipListParams,
  DropshipListResponse,
  DropshipOrderStatus,
} from "@/types/dropship";

// ─── 드롭십 상품 ──────────────────────────────────────────────────────────────

/** GET /dropship/products — 도매 드롭십 상품 목록 */
export async function getDropshipProducts(
  params?: DropshipListParams
): Promise<DropshipListResponse<DropshipProduct>> {
  const search = new URLSearchParams();
  if (params?.status) search.set("status", params.status);
  if (params?.page != null) search.set("page", String(params.page));
  if (params?.per_page != null) search.set("per_page", String(params.per_page));
  if (params?.search) search.set("search", params.search);
  if (params?.from) search.set("from", params.from);
  if (params?.to) search.set("to", params.to);
  const qs = search.toString();
  const res = await fetchApi<Partial<DropshipListResponse<DropshipProduct>>>(
    `dropship/products${qs ? `?${qs}` : ""}`
  );
  return {
    data: res?.data ?? [],
    current_page: res?.current_page ?? 1,
    last_page: res?.last_page ?? 1,
    per_page: res?.per_page ?? 15,
    total: res?.total ?? 0,
  };
}

/** GET /dropship/products/{id} — 드롭십 상품 상세 */
export async function getDropshipProduct(id: number): Promise<DropshipProduct> {
  const res = await fetchApi<DropshipProduct>(`dropship/products/${id}`);
  return res;
}

// ─── 드롭십 신청 ──────────────────────────────────────────────────────────────

/** POST /dropship/apply/{productId} — 드롭십 신청 */
export async function applyDropship(
  productId: number
): Promise<DropshipApplication> {
  const res = await fetchApi<DropshipApplication>(`dropship/apply/${productId}`, {
    method: "POST",
    body: JSON.stringify({}),
  });
  return res;
}

// ─── 드롭십 주문 ──────────────────────────────────────────────────────────────

/** GET /dropship — 드롭십 주문 목록 */
export async function getDropshipOrders(
  params?: DropshipListParams
): Promise<DropshipListResponse<DropshipOrder>> {
  const search = new URLSearchParams();
  if (params?.status) search.set("status", params.status);
  if (params?.page != null) search.set("page", String(params.page));
  if (params?.per_page != null) search.set("per_page", String(params.per_page));
  if (params?.search) search.set("search", params.search);
  if (params?.from) search.set("from", params.from);
  if (params?.to) search.set("to", params.to);
  const qs = search.toString();
  const res = await fetchApi<Partial<DropshipListResponse<DropshipOrder>>>(
    `dropship${qs ? `?${qs}` : ""}`
  );
  return {
    data: res?.data ?? [],
    current_page: res?.current_page ?? 1,
    last_page: res?.last_page ?? 1,
    per_page: res?.per_page ?? 15,
    total: res?.total ?? 0,
  };
}

/** GET /dropship/{id} — 드롭십 주문 상세 */
export async function getDropshipOrder(id: number): Promise<DropshipOrder> {
  const res = await fetchApi<DropshipOrder>(`dropship/${id}`);
  return res;
}

/** PUT /dropship/{id}/status — 드롭십 주문 상태 변경 */
export async function updateDropshipOrderStatus(
  id: number,
  status: DropshipOrderStatus

  const res = await fetchApi<DropshipOrder>(`dropship/${id}/status`, {
    method: "PUT",
    body: JSON.stringify({ status }),

  return res;
}
