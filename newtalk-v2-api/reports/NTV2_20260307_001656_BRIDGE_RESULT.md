**최종 파일 내용:**
```php
<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\JsonResponse;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Redis;

/**
 * Health Check API
 * R5-API-HEALTH-001: GET /api/health — 서비스 모니터링 엔드포인트 (인증 불필요)
 */
class HealthController extends Controller
{
    /**
     * GET /api/health
     * DB, Redis, 디스크 상태를 확인하고 서비스 헬스를 반환.
     *
     * 응답: { status: "ok"|"degraded", timestamp: "...", services: { database: "ok"|"error", redis: "ok"|"error", disk_free_gb: N } }
     * HTTP 200 (정상) 또는 503 (장애)
     */
    public function index(): JsonResponse
    {
        $services = [];
        $degraded = false;

        // DB 연결 확인
        try {
            DB::connection()->getPdo();
            DB::select('SELECT 1');
            $services['database'] = 'ok';
        } catch (\Throwable $e) {
            $services['database'] = 'error';
            $degraded = true;
        }

        // Redis 연결 확인
        try {
            Redis::ping();
            $services['redis'] = 'ok';
        } catch (\Throwable $e) {
            $services['redis'] = 'error';
            $degraded = true;
        }

        // 디스크 여유 공간 (GB)
        try {
            $diskFreeBytes = disk_free_space('/');
            $services['disk_free_gb'] = round($diskFreeBytes / (1024 ** 3), 2);
        } catch (\Throwable $e) {
            $services['disk_free_gb'] = null;
        }

        $status = $degraded ? 'degraded' : 'ok';
        $httpCode = $degraded ? 503 : 200;

        return response()->json([
            'status'    => $status,
            'timestamp' => now()->toIso8601String(),
            'services'  => $services,
        ], $httpCode);
    }

    public function full()
    {
        $r = [];
        $r["infra"] = ["php" => PHP_VERSION, "laravel" => app()->version(), "time" => now()->toIso8601String(), "disk_free_gb" => round(disk_free_space("/") / 1073741824, 2)];
        try { $t = \DB::select("SHOW TABLES"); $r["database"] = ["status" => "ok", "table_count" => count($t), "user_count" => \DB::table("users")->count(), "product_count" => \DB::table("products")->count()]; } catch (\Exception $e) { $r["database"] = ["status" => "error"]; }
        try { \Illuminate\Support\Facades\Redis::ping(); $r["redis"] = ["status" => "ok"]; } catch (\Exception $e) { $r["redis"] = ["status" => "error"]; }
        $r["routes"] = ["total" => count(app("router")->getRoutes()->getRoutes())];
        foreach (["admin@newtalk.kr","md@newtalk.kr","purchaser@newtalk.kr","wholesale@newtalk.kr","retail@newtalk.kr","outsource@newtalk.kr"] as $em) { $r["users"][$em] = \DB::table("users")->where("email",$em)->exists() ? "ok" : "missing"; }
        $b = base_path(); $r["git"] = ["branch" => trim(shell_exec("cd $b && git rev-parse --abbrev-ref HEAD 2>/dev/null") ?: "?"), "sha" => trim(shell_exec("cd $b && git rev-parse --short HEAD 2>/dev/null") ?: "?")];
        $bf = base_path("frontend/.next/BUILD_ID"); $r["frontend"] = ["build_exists" => file_exists($bf)];
        $fp = @fsockopen("127.0.0.1", 6001, $en, $es, 2); $r["reverb"] = ["port_6001" => $fp ? "listening" : "closed"]; if ($fp) fclose($fp);
        return response()->json($r);
    }
}
```
