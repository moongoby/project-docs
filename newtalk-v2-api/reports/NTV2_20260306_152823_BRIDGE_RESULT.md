### MessengerLayout.tsx
```typescript
"use client";

import { useState } from "react";
import { MessageSquare } from "lucide-react";
import type { Conversation } from "@/types/messenger";
import { ConversationList } from "./ConversationList";
import { MessageView } from "./MessageView";

function getConversationTitle(conv: Conversation): string {
  return (
    conv.title ??
    conv.participants?.[0]?.user?.name ??
    `대화 #${conv.id}`
  );
}

export function MessengerLayout() {
  const [selectedConversation, setSelectedConversation] =
    useState<Conversation | null>(null);
  const [mobileView, setMobileView] = useState<"list" | "chat">("list");

  const handleSelect = (conv: Conversation) => {
    setSelectedConversation(conv);
    setMobileView("chat");
  };

  const handleBack = () => {
    setMobileView("list");
  };

  return (
    <div className="flex h-[calc(100vh-4rem)] overflow-hidden rounded-xl border border-zinc-200 bg-white shadow-sm">
      <div className={[
        "w-full md:w-80 md:flex md:flex-col md:shrink-0",
        mobileView === "list" ? "flex flex-col" : "hidden",
      ].join(" ")}>
        <ConversationList
          selectedId={selectedConversation?.id}
          onSelect={handleSelect}
        />
      </div>
      <div className={[
        "flex-1 flex flex-col",
        mobileView === "chat" ? "flex" : "hidden md:flex",
      ].join(" ")}>
        {selectedConversation ? (
          <MessageView
            key={selectedConversation.id}
            conversationId={selectedConversation.id}
            conversationTitle={getConversationTitle(selectedConversation)}
            onBack={handleBack}
          />
        ) : (
          <div className="flex flex-1 flex-col items-center justify-center gap-3 text-zinc-400">
            <MessageSquare className="h-12 w-12 opacity-30" />
            <p className="text-sm">대화를 선택하세요</p>
          </div>
        )}
      </div>
    </div>
  );
}
```
