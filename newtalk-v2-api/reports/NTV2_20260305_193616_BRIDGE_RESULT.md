## Step 3 — settlement-api.ts 생성

```typescript
import { fetchApi } from "./api";
import type {
  Settlement,
  SettlementItem,
  SettlementLog,
  SettlementCreateRequest,
  SettlementListParams,
  SettlementListResponse,
} from "@/types/settlement";

/** GET /api/settlements — 정산 목록 (페이지네이션, admin: 전체 / wholesale: 본인) */
export async function getSettlements(
  params?: SettlementListParams
): Promise<SettlementListResponse> {
  const search = new URLSearchParams();
  if (params?.page != null) search.set("page", String(params.page));
  if (params?.per_page != null) search.set("per_page", String(params.per_page));
  if (params?.status) search.set("status", params.status);
  if (params?.seller_id != null) search.set("seller_id", String(params.seller_id));
  if (params?.period_start) search.set("period_start", params.period_start);
  if (params?.period_end) search.set("period_end", params.period_end);
  const qs = search.toString();
  const res = await fetchApi<SettlementListResponse>(`settlements${qs ? `?${qs}` : ""}`);
  return res as unknown as SettlementListResponse;
}

/** GET /api/settlements/{id} — 정산 상세 (응답: { data: Settlement }) */
export async function getSettlement(id: number): Promise<Settlement> {
  const res = await fetchApi<{ data: Settlement }>(`settlements/${id}`);
  return (res as { data: Settlement }).data;
}

/** POST /api/settlements — 정산 생성 (admin) */
export async function createSettlement(
  data: SettlementCreateRequest
): Promise<Settlement> {
  const res = await fetchApi<{ message: string; data: Settlement }>(
    "settlements",
    {
      method: "POST",
      body: JSON.stringify(data),
    }
  );
  return (res as { data: Settlement }).data;
}

/** PUT /api/settlements/{id}/confirm — 정산 확정 (admin) */
export async function confirmSettlement(id: number): Promise<Settlement> {
  const res = await fetchApi<{ message: string; data: Settlement }>(
    `settlements/${id}/confirm`,
    { method: "PUT" }
  );
  return (res as { data: Settlement }).data;
}

/** GET /api/settlements/{id}/items — 정산 항목 목록 */
export async function getSettlementItems(id: number): Promise<SettlementItem[]> {
  const res = await fetchApi<{ data: SettlementItem[] }>(`settlements/${id}/items`);
  return (res as { data: SettlementItem[] }).data ?? [];
}

/** GET /api/settlements/{id}/logs — 정산 로그 목록 */
export async function getSettlementLogs(id: number): Promise<SettlementLog[]> {
  const res = await fetchApi<{ data: SettlementLog[] }>(`settlements/${id}/logs`);
  return (res as { data: SettlementLog[] }).data ?? [];
}
```

**함수 수**: 6개 (getSettlements, getSettlement, createSettlement, confirmSettlement, getSettlementItems, getSettlementLogs)

/** 정산 상태 */
export type SettlementStatus = "pending" | "confirmed" | "paid" | "disputed";

export interface SettlementSeller {
  id: number;
  name: string;
  email: string;
}

export interface SettlementItem {
  id: number;
  settlement_id: number;
  order_id: number | null;
  description: string;
  amount: number;
  created_at: string;
}

export interface SettlementLog {
  id: number;
  settlement_id: number;
  action: string;
  note: string | null;
  created_at: string;
}

export interface Settlement {
  id: number;
  seller_id: number;
  seller?: SettlementSeller;
  status: SettlementStatus;
  period_start: string;
  period_end: string;
  total_amount: number;
  confirmed_at?: string | null;
  paid_at?: string | null;
  note?: string | null;
  items?: SettlementItem[];
  logs?: SettlementLog[];
  created_at: string;
  updated_at: string;
}

export interface SettlementSummary {
  total_count: number;
  pending_count: number;
  confirmed_count: number;
  paid_count: number;
  disputed_count: number;
  total_amount: number;
  pending_amount: number;
  confirmed_amount: number;
  paid_amount: number;
}

export interface SettlementCreateRequest {
  seller_id: number;
  period_start: string;
  period_end: string;
  total_amount: number;
  note?: string;
}

export interface SettlementListParams {
  page?: number;
  per_page?: number;
  status?: string;
  seller_id?: number;
  period_start?: string;

}

export interface SettlementListResponse {

  total: number;
  from?: number;
  to?: number;
